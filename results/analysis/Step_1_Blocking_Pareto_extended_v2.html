Blocking-method comparison at target recall levels¶

This notebook evaluates four blocking methods across six datasets:

  • LSH
  • LSH+BP (LSH + Block Purging)
  • LSH+BF (LSH + Block Filtering)
  • LSH+BP+BF (LSH + Block Purging + Block Filtering)

For each dataset and target recall level (85%, 90%, 95%), exactly one configuration is selected for each method:

  1. Find all configurations with Recall % >= target.
  2. If at least one exists, select the configuration with the highest precision.
  3. If none exists, select the configuration with the highest recall and mark it as a fallback.
  4. Break ties using the other metric: precision first/recall second when the target is met; recall first/precision second for fallback selections.

The final tables report precision/recall, candidate pairs, and runtime for every selected method configuration.

def get_method_file(method, dataset):
    """Return the CSV path for a method and dataset."""
    return MAIN_DIR / METHOD_FILES[method].as_posix().format(dataset=dataset)


def load_results():
    """Load all result CSVs into a dictionary keyed by dataset and method."""
    data = {}
    missing = []

    for dataset in DATASETS:
        data[dataset] = {}

        for method in METHOD_ORDER:
            path = get_method_file(method, dataset)

            if not path.exists():
                missing.append(str(path))
                continue

            df = pd.read_csv(path)

            # Make sure the metrics used for selection are numeric.
            for col in ['Precision %', 'Recall %', 'F1 %', 'runtime', 'candidate_pairs']:
                if col in df.columns:
                    df[col] = pd.to_numeric(df[col], errors='coerce')

            data[dataset][method] = df

    if missing:
        print('WARNING: The following result files were not found:')
        for path in missing:
            print('  ', path)

    return data

# ============================================================
# 3. Configuration-selection rule
# ============================================================

def select_configuration(df, threshold):
    """
    Select exactly one configuration according to the requested rule.

    Rule:
      1. Keep configurations with Recall % >= threshold.
      2. If any exist, choose highest precision, then highest recall.
      3. Otherwise choose highest recall, then highest precision,
         and mark the result as a fallback.
    """

    required = ['Precision %', 'Recall %']
    missing_cols = [c for c in required if c not in df.columns]
    if missing_cols:
        raise ValueError(f'Missing required columns: {missing_cols}')

    candidates = df.dropna(subset=required).copy()

    if candidates.empty:
        return None

    # First choice: configurations that meet the requested recall.
    eligible = candidates[candidates['Recall %'] >= threshold]

    if not eligible.empty:
        selected = (
            eligible
            .sort_values(
                by=['Precision %', 'Recall %'],
                ascending=[False, False],
                kind='mergesort'
            )
            .iloc[0]
        )
        selection_type = 'Threshold met'
        fallback = False
    else:
        # No configuration reaches the target: maximize recall,
        # using precision as the secondary criterion.
        selected = (
            candidates
            .sort_values(
                by=['Recall %', 'Precision %'],
                ascending=[False, False],
                kind='mergesort'
            )
            .iloc[0]
        )
        selection_type = 'Fallback: max recall'
        fallback = True

    return {
        'row': selected,
        'selection_type': selection_type,
        'fallback': fallback,
    }

def build_selected_results(results):
    rows = []

    for dataset in DATASETS:
        for threshold in THRESHOLDS:
            for method in METHOD_ORDER:

                df = results.get(dataset, {}).get(method)

                if df is None:
                    rows.append({
                        'Dataset': dataset,
                        'Target Recall': threshold,
                        'Method': method,
                        'Selection': 'No result file',
                        'Fallback': True,
                        'Precision %': np.nan,
                        'Recall %': np.nan,
                        'Candidate pairs': np.nan,
                        'Runtime': np.nan,
                        'F1 %': np.nan,
                        'Configuration': None,
                        'Ratio': np.nan,
                    })
                    continue

                selection = select_configuration(df, threshold)

                if selection is None:
                    rows.append({
                        'Dataset': dataset,
                        'Target Recall': threshold,
                        'Method': method,
                        'Selection': 'No valid configuration',
                        'Fallback': True,
                        'Precision %': np.nan,
                        'Recall %': np.nan,
                        'Candidate pairs': np.nan,
                        'Runtime': np.nan,
                        'F1 %': np.nan,
                        'Configuration': None,
                        'Ratio': np.nan,
                    })
                    continue

                row = selection['row']

                rows.append({
                    'Dataset': dataset,
                    'Target Recall': threshold,
                    'Method': method,
                    'Selection': selection['selection_type'],
                    'Fallback': selection['fallback'],
                    'Precision %': row.get('Precision %', np.nan),
                    'Recall %': row.get('Recall %', np.nan),
                    'Candidate pairs': row.get('candidate_pairs', np.nan),
                    'Runtime': row.get('runtime', np.nan),
                    'F1 %': row.get('F1 %', np.nan),
                    'Configuration': row.get('config', None),
                    'Ratio': row.get('ratio', np.nan),
                })

    out = pd.DataFrame(rows)
                
    out['Method'] = pd.Categorical(
        out['Method'], categories=METHOD_ORDER, ordered=True
    )
    out['Dataset'] = pd.Categorical(
        out['Dataset'], categories=DATASETS, ordered=True
    )

    return out.sort_values(
        ['Target Recall', 'Dataset', 'Method']
    ).reset_index(drop=True)

# 5. Create the requested table format
# ============================================================

def format_number(value, decimals=2):
    if pd.isna(value):
        return '—'
    return f'{value:,.{decimals}f}'


def format_integer(value):
    if pd.isna(value):
        return '—'
    return f'{int(round(value)):,}'


def make_threshold_table(selected_results, threshold):
    """
    Create one table for a target recall threshold.

    Each method has four separate columns:
      Recall | Precision | Candidate pairs | Runtime

    A dagger (†) marks a fallback configuration, i.e. no configuration
    reached the target recall and the maximum-recall configuration was used.
    """

    subset = selected_results[
        selected_results['Target Recall'] == threshold
    ].copy()

    html = []
    html.append('<style>')
    html.append('.blocking-table { border-collapse: collapse; width: 100%; margin: 12px 0 8px 0; }')
    html.append('.blocking-table th, .blocking-table td { border: 1px solid #bdbdbd; padding: 7px 9px; text-align: center; vertical-align: middle; }')
    html.append('.blocking-table thead tr:first-child th { background: #e8e8e8; font-weight: 700; }')
    html.append('.blocking-table thead tr:nth-child(2) th { background: #f4f4f4; font-weight: 600; }')
    html.append('.blocking-table td:first-child { font-weight: 600; text-align: left; }')
    html.append('.fallback { background: #fff3cd; }')
    html.append('.note { font-size: 0.9em; color: #555; margin-bottom: 18px; }')
    html.append('</style>')

    html.append(f'<h3>Target recall {threshold}%</h3>')
    html.append('<table class="blocking-table">')

    # First header row: method names spanning four metric columns.
    html.append('<thead><tr>')
    html.append('<th rowspan="2">Dataset</th>')
    for method in METHOD_ORDER:
        html.append(f'<th colspan="4">{method}</th>')
    html.append('</tr>')

    # Second header row: four metrics for every method.
    html.append('<tr>')
    for _ in METHOD_ORDER:
        html.append('<th>Recall (%)</th>')
        html.append('<th>Precision (%)</th>')
        html.append('<th>Candidate pairs</th>')
        html.append('<th>Runtime</th>')
    html.append('</tr></thead><tbody>')

    for dataset in DATASETS:
        html.append('<tr>')
        html.append(f'<td>{dataset}</td>')

        for method in METHOD_ORDER:
            match = subset[
                (subset['Dataset'] == dataset) &
                (subset['Method'] == method)
            ]

            if match.empty:
                html.extend(['<td>—</td>'] * 4)
                continue

            row = match.iloc[0]
            css_class = ' class="fallback"' if row['Fallback'] else ''
            marker = ' †' if row['Fallback'] else ''

            html.append(f'<td{css_class}>{format_number(row["Recall %"])}{marker}</td>')
            html.append(f'<td{css_class}>{format_number(row["Precision %"])}{marker}</td>')
            html.append(f'<td{css_class}>{format_integer(row["Candidate pairs"])}</td>')
            html.append(f'<td{css_class}>{format_number(row["Runtime"])}</td>')

        html.append('</tr>')

    html.append('</tbody></table>')
    html.append('<div class="note">† = fallback: no configuration reached the target recall; the configuration with maximum recall was selected. Runtime is reported in the units stored in the input CSV files.</div>')

    return ''.join(html)

Interpretation of the final tables¶

For each recall threshold, every blocking method has four separate columns:

  • Recall (%): recall of the selected configuration;
  • Precision (%): precision of the selected configuration;
  • Candidate pairs: number of candidate pairs generated by the selected configuration;
  • Runtime: runtime of the selected configuration, in the units stored in the input CSV files.

A † marks a fallback selection. This means that no configuration for that method achieved the requested recall threshold, so the configuration with the highest attainable recall was selected instead.

The configuration-inspection section below displays the complete set of selected configurations directly as a table in the notebook.

Pareto-front analysis across the six datasets¶

For each target recall level (85%, 90%, and 95%), this section uses the selected configuration for each method and dataset from the preceding analysis. It then computes the average precision and average recall of each blocking method across D2, D3, D4, D8, D9, and D10.

The Pareto front is defined with respect to maximizing both average precision and average recall. A method is dominated if another method has at least as high average precision and recall, with at least one of the two strictly higher.

# ============================================================
# 10. Identify the Pareto front
# ============================================================

def pareto_front(df, x_col="Average_Recall", y_col="Average_Precision"):
    """Return rows that are non-dominated when maximizing x and y."""
    points = df.dropna(subset=[x_col, y_col]).copy()
    keep = []
    values = points[[x_col, y_col]].to_numpy()

    for i, (x_i, y_i) in enumerate(values):
        dominated = False
        for j, (x_j, y_j) in enumerate(values):
            if i == j:
                continue
            if (x_j >= x_i and y_j >= y_i) and (x_j > x_i or y_j > y_i):
                dominated = True
                break
        keep.append(not dominated)

    return points.loc[keep].copy()
